       tg.ACTION_TIMING, tg.ACTION_STATEMENT
FROM information_schema.`TRIGGERS` tg
WHERE tg.TRIGGER_SCHEMA = 'base_teste_pl';

DESC itemnotafiscal;
DESC medicamento;

-- criando trigger para atualizar dados da venda de medicamento
delimiter $$ -- ajuste do limitar final de criação do gatinho
CREATE OR REPLACE TRIGGER tg_itemnotafiscal_venda AFTER INSERT
    ON itemnotafiscal FOR EACH ROW
BEGIN
    UPDATE medicamento m SET m.qt_estoque = m.qt_estoque - NEW.qt_vendida
    WHERE m.cd_medicamento = NEW.cd_medicamento;
END $$

    ON itemnotafiscal FOR EACH ROW
BEGIN
    DECLARE var_estoque INT DEFAULT 0;
    
    SELECT m.qt_estoque INTO var_estoque 
    FROM medicamento m 
    WHERE m.cd_medicamento = NEW.cd_medicamento;

    -- ou
    SET var_estoque = (
        SELECT m.qt_estoque 
        FROM medicamento m 
        WHERE m.cd_medicamento = NEW.cd_medicamento
    );

    IF var_estoque < NEW.qt_vendida THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Estoque insuficiente!';
    END IF;

    UPDATE medicamento m 
    SET m.qt_estoque = m.qt_estoque - NEW.qt_vendida
    WHERE m.cd_medicamento = NEW.cd_medicamento;
END $$

delimiter $$ -- ajiste do limitar final de criacao do gatilho
CREATE OR REPLACE TRIGGER tg_itemnotafiscal_venda BEFORE INSERT 
				ON itemnotafiscal FOR EACH ROW 
BEGIN 
	DECLARE var_estoque INT DEFAULT 0; -- declaracao da variavel que recebera o estoque do medicamento
	DECLARE var_medicamento DECIMAL (8,2) DEFAULT 0.00; -- armazena preço venda do medicamento
	SELECT m.qt_estoque, m.vl_venda INTO var_estoque, var_medicamento	
		 FROM medicamento m WHERE m.cd_medicamento = NEW.cd_medicamento;
	SET NEW.vl_venda = var_medicamento; -- atualiza o valor do medicamento na venda (itemnotafiscal)
	IF var_estoque < NEW.qt_vendida THEN
		SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Estoque insuficiente!';
	END IF;
	UPDATE medicamento m SET m.qt_estoque = m.qt_estoque - NEW.qt_vendida
		WHERE m.cd_medicamento = NEW.cd_medicamento;
	-- atualizaa o valor total da nota fiscal
	UPDATE notafiscal nf SET nf.vl_total = nf.vl_total + (NEW.qt_vendida * var_medicamento)
		WHERE nf.nr_notafiscal = NEW.nr_notafiscal;
END $$
