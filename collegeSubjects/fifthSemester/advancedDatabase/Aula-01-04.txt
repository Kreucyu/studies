CREATE TABLE escolaridade
(id_escolaridade INT AUTO_INCREMENT,

PRIMARY KEY(id_escolaridade)
);

	SELECT DISTINCT(p.escolaridade)
	FROM pessoa p
	ORDER BY 1;
